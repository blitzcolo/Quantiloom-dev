4. 实现 Planck 黑体辐射函数

// Planck's law: spectral radiance of blackbody at temperature T and wavelength λ
// Returns: L_λ(T, λ) in W·sr⁻¹·m⁻²·nm⁻¹
//
// Formula:
//   L_λ(T, λ) = (2hc²/λ⁵) / (exp(hc/λkT) - 1)
//
// Constants:
//   h = 6.62607015e-34 J·s (Planck constant)
//   c = 299792458 m/s (speed of light)
//   k = 1.380649e-23 J/K (Boltzmann constant)
//
// Units:
//   T: Kelvin
//   λ: meters (convert from nm: λ_m = λ_nm * 1e-9)
//
float PlanckRadiance(float temperature_K, float wavelength_nm) {
    const float h = 6.62607015e-34;  // Planck constant (J·s)
    const float c = 299792458.0;     // Speed of light (m/s)
    const float k = 1.380649e-23;    // Boltzmann constant (J/K)
    
    float lambda_m = wavelength_nm * 1e-9;  // Convert nm to meters
    
    // Compute constants
    float c1 = 2.0 * h * c * c;  // 2hc²
    float c2 = h * c / k;        // hc/k
    
    // Planck's law
    float numerator = c1 / pow(lambda_m, 5.0);
    float exponent = c2 / (lambda_m * temperature_K);
    float denominator = exp(exponent) - 1.0;
    
    // L_λ in W·sr⁻¹·m⁻²·m⁻¹, convert to W·sr⁻¹·m⁻²·nm⁻¹
    float L_lambda = numerator / denominator;
    return L_lambda * 1e-9;  // Convert /m to /nm
}

void RayTracingPipeline::BindTangentBuffer(const GpuBuffer& buffer) {
    VkDevice device = m_context.GetDevice();
    
    VkDescriptorBufferInfo bufferInfo{};
    bufferInfo.buffer = buffer.GetHandle();
    bufferInfo.offset = 0;
    bufferInfo.range = VK_WHOLE_SIZE;
    
    VkWriteDescriptorSet write{};
    write.sType = VK_STRUCTURE_TYPE_WRITE_DESCRIPTOR_SET;
    write.dstSet = m_descriptorSet;
    write.dstBinding = 9;
    write.dstArrayElement = 0;
    write.descriptorType = VK_DESCRIPTOR_TYPE_STORAGE_BUFFER;
    write.descriptorCount = 1;
    write.pBufferInfo = &bufferInfo;
    
    vkUpdateDescriptorSets(device, 1, &write, 0, nullptr);
    QL_LOG_DEBUG("  [DEBUG] Bound tangent buffer to binding 9");
}
